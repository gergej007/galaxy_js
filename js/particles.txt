<canvas id="particleCanvas" width="500" height="75"></canvas>
<script>
  const canvas = document.getElementById('particleCanvas');
  const ctx = canvas.getContext('2d');

  const width = canvas.width;
  const height = canvas.height;
  const centerX = width / 2;
  const centerY = height / 2;
  const numParticles = 80;
  const particlesPerSide = 20;
  const duration = 1000; // 1 second
  const fadeOutDuration = 50; // 50 ms
  const fps = 15;
  const totalFrames = fps * (duration / 1000);
  const animationFrames = Math.floor(fps * ((duration - fadeOutDuration) / 1000));
  const fadeFrames = totalFrames - animationFrames;
  const startSize = 1;
  const endSize = 3;

  let particles = [];

  function initParticles() {
    for (let i = 0; i < numParticles; i++) {
      let angle;
      // Assign direction based on index
      if (i < particlesPerSide) angle = Math.random() * Math.PI / 2 - Math.PI / 4; // Right
      else if (i < particlesPerSide * 2) angle = Math.random() * Math.PI / 2 + 3 * Math.PI / 4; // Left
      else if (i < particlesPerSide * 3) angle = Math.random() * Math.PI / 2 + Math.PI / 4; // Bottom
      else angle = Math.random() * Math.PI / 2 - 3 * Math.PI / 4; // Top

      const speed = Math.random() * 2 + 1;
      particles.push({
        x: centerX,
        y: centerY,
        vx: Math.cos(angle) * speed,
        vy: Math.sin(angle) * speed,
        size: startSize,
        opacity: 1
      });
    }
  }

  function draw() {
    ctx.clearRect(0, 0, width, height); // Clear canvas for transparent background
    ctx.fillStyle = 'orange';

    particles.forEach(p => {
      ctx.globalAlpha = p.opacity;
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.size / 2, 0, Math.PI * 2);
      ctx.fill();
    });
  }

  function update(frame) {
    if (frame <= animationFrames) {
      // Explosion phase
      const progress = frame / animationFrames;
      particles.forEach(p => {
        p.x += p.vx;
        p.y += p.vy;
        p.size = startSize + (endSize - startSize) * progress;
      });
    } else if (frame <= totalFrames) {
      // Fade out phase
      const fadeProgress = (frame - animationFrames) / fadeFrames;
      particles.forEach(p => {
        p.opacity = 1 - fadeProgress;
      });
    }
  }

  let currentFrame = 0;
  initParticles();

  const interval = setInterval(() => {
    draw();
    update(currentFrame);
    currentFrame++;

    if (currentFrame > totalFrames) {
      clearInterval(interval);
      // To create a GIF, you would typically capture these frames using a library like GIF.js
      // and then download the resulting blob. For this demonstration, we run the animation once.
      console.log('Animation finished.');
    }
  }, 1000 / fps);

</script>
